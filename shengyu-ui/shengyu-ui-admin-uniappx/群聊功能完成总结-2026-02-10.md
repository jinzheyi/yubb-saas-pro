function generateGroupName(members: any[]): string {
  if (members.length === 0) return '群聊'
  if (members.length <= 3) {
    return members.map(m => m.name).join('、')
  }
  const firstThree = members.slice(0, 3).map(m => m.name).join('、')
  return `${firstThree}等${members.length}人`
}
```

// 状态变量
const chatType = ref('single')  // 'single' | 'group'
const groupId = ref(0)
const memberCount = ref(0)

// 页面加载
onLoad((options: any) => {
  if (options['type']) chatType.value = options['type']
  if (options['groupId']) groupId.value = parseInt(options['groupId'])
  if (options['memberCount']) memberCount.value = parseInt(options['memberCount'])
  
  if (chatType.value === 'group') {
    loadGroupMessages()
  }
})

// 头部显示
<text v-if="chatType === 'single'" class="header-status">手机在线</text>
<text v-if="chatType === 'group'" class="header-status">{{ memberCount }}人</text>
```

## 后端 API 对接准备

### 需要对接的 API

1. **创建群聊 API**
   ```
   POST /api/group/create
   ```
   - 请求参数：groupName, groupType, memberIds, settings
   - 响应数据：groupId, groupName, memberCount, createTime

2. **获取群聊消息 API**
   ```
   GET /api/group/messages?groupId=xxx&page=1&pageSize=20
   ```
   - 请求参数：groupId, page, pageSize
   - 响应数据：list, total, hasMore

3. **发送群聊消息 API**
   ```
   POST /api/group/message/send
   ```
   - 请求参数：groupId, type, content
   - 响应数据：messageId, timestamp
