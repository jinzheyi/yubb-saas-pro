## 核心技术实现

### 1. 状态同步机制

```typescript
// 同步本地选中状态与全局状态
function syncSelectionState() {
  for (let i = 0; i < memberList.value.length; i++) {
    memberList.value[i].selected = isMemberSelected(memberList.value[i].id)
  }
}

// 页面加载时同步
onLoad((options: any) => {
  if (options != null && options['mode'] === 'select') {
    isSelectMode.value = true
  }
  syncSelectionState()
})

// 页面显示时同步（从子页面返回时）
onShow(() => {
  if (isSelectMode.value) {
    syncSelectionState()
  }
})
```

### 2. 选择操作

```typescript
// 切换成员选中状态
function toggleMemberSelect(member: any) {
  toggleMember(member)  // 更新全局状态
  member.selected = isMemberSelected(member.id)  // 同步本地状态
}
```

### 3. 层级选择（组织结构）

```typescript
// 批量添加成员
function toggleOrgSelect(org: any) {
  org.selected = !org.selected
  
  const allMembers: any[] = []
  // 收集所有成员...
  
  if (org.selected) {
    addMembers(allMembers)  // 批量添加到全局状态
  } else {
    const memberIds = allMembers.map(m => m.id)
    removeMembers(memberIds)  // 批量移除
  }
}
```

### 4. 生命周期管理

```typescript
// 主入口页面
onLoad(() => {
  startSelection()  // 开始选择模式
})

onUnload(() => {
  endSelection()  // 清空所有数据
})
```
