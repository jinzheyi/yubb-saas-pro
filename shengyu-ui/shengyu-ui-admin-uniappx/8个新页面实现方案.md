**文件路径：** `pages/message/group-members.uvue`

**功能特性：**
- 搜索框
- 按字母分组
- 索引导航
- 成员详情跳转
- 成员数量统计

**页面结构：**
```vue
<template>
  <view class="page">
    <!-- 头部 -->
    <view class="header">
      <view class="header-left" @click="handleBack">...</view>
      <text class="header-title">群成员({{ memberCount }})</text>
      <view class="header-right"></view>
    </view>

    <view class="search-wrap">...</view>

      <view class="letter-group" v-for="group in memberGroups" :id="'letter-' + group.letter">
        <text class="letter-title">{{ group.letter }}</text>
        
        <!-- 成员项 -->
        <view class="member-item" v-for="member in group.members" @click="handleMemberClick(member)">
          <view class="member-avatar" :style="{ backgroundColor: member.avatarBg }">
            <text class="member-avatar-text">{{ member.avatarText }}</text>
          </view>
          <view class="member-info">
            <text class="member-name">{{ member.name }}</text>
            <text class="member-role" v-if="member.role">{{ member.role }}</text>
          </view>
        </view>
      </view>
    </scroll-view>
    
    <!-- 字母索引 -->
    <view class="letter-index">
      <text class="index-letter" v-for="letter in letters" @click="scrollToLetter(letter)">
        {{ letter }}
      </text>
    </view>
  </view>
</template>
```

**数据处理：**
```typescript
// 按首字母分组
function groupMembersByLetter(members: any[]) {
  const groups = {}
  members.forEach(member => {
    const letter = member.pinyin[0].toUpperCase()
    if (!groups[letter]) {
      groups[letter] = []
    }
    groups[letter].push(member)
  })
  return Object.keys(groups).sort().map(letter => ({
    letter,
    members: groups[letter]
  }))
}
```
